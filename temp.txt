        private static async Task ProccessMail()
        {
            var config = new ConfigurationBuilder()
    .AddJsonFile("appsettings.json")
    .Build();


var mail = config.GetSection("Mail");


string imapServer = mail["ImapServer"]!;
int imapPort = int.Parse(mail["ImapPort"]!);

string smtpServer = mail["SmtpServer"]!;
int smtpPort = int.Parse(mail["SmtpPort"]!);

string login = mail["Login"]!;
string password = mail["Password"]!;


Console.WriteLine("Memo.AI Mail Listener started");



using var client = new ImapClient();


client.Connect(
    imapServer,
    imapPort,
    true);


client.Authenticate(
    login,
    password);


var inbox = client.Inbox;

inbox.Open(
    MailKit.FolderAccess.ReadWrite);


Console.WriteLine(
    $"Писем в ящике: {inbox.Count}");



var uids = inbox.Search(
    SearchQuery.NotSeen);



foreach(var uid in uids)
{
    var message = inbox.GetMessage(uid);


    Console.WriteLine("----------------");
    Console.WriteLine(
        $"От: {message.From}");

    Console.WriteLine(
        $"Тема: {message.Subject}");

    Console.WriteLine(
        message.TextBody);


    await Reply(
        message,
        smtpServer,
        smtpPort,
        login,
        password);


    inbox.AddFlags(
        uid,
        MessageFlags.Seen,
        true);
}



client.Disconnect(true);

Console.WriteLine("Готово");
        }
        static async Task Reply(
    MimeMessage original,
    string smtpServer,
    int smtpPort,
    string login,
    string password)
{

    var reply = new MimeMessage();


    reply.From.Add(
        MailboxAddress.Parse(login));


    reply.To.Add(
        original.From.Mailboxes.First());


    reply.Subject =
        "Re123: " + original.Subject;



    reply.InReplyTo =
        original.MessageId;



   if (!string.IsNullOrEmpty(original.MessageId))
{
    reply.InReplyTo = original.MessageId;

    reply.References.Add(
        original.MessageId);
}



    reply.Body =
        new TextPart("plain")
        {
            Text =
            """
            Принято.

            Ваша мысль получена и будет обработана AI-агентами.
            """
        };



    using var smtp = new SmtpClient();


    await smtp.ConnectAsync(
        smtpServer,
        smtpPort,
        true);


    await smtp.AuthenticateAsync(
        login,
        password);


    await smtp.SendAsync(reply);


    await smtp.DisconnectAsync(true);


    Console.WriteLine(
        "Ответ отправлен");
}
